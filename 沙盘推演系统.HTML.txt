        function generateTimeSeries(sku, combinedMktRate, actualRev, tShift) {
            if(!tShift) tShift = 1.0;
            var months = 36; var salesBase = [];
            for(var m=1; m<=months; m++) {
                var effM = m / tShift; 
                if(sku.stage === 'intro') salesBase.push(Math.pow(effM/18, 2) * Math.exp(-effM/9));
                else if(sku.stage === 'growth') salesBase.push(Math.pow((effM+12)/18, 2) * Math.exp(-(effM+12)/9));
                else if(sku.stage === 'mat') salesBase.push(Math.pow((effM+24)/18, 2) * Math.exp(-(effM+24)/9));
                else salesBase.push(Math.exp(-effM/12));
            }
            var sum12 = 0; for(var i=0; i<12; i++) sum12 += salesBase[i];
            var multiplier = sum12 > 0 ? (actualRev / sum12) : 0; 
            var cashFlow = []; var cumulativeCash = -sku.sunkCost; 
            var totalMktBudget = 0; for(var k=0; k<months; k++) totalMktBudget += (salesBase[k] * multiplier);
            totalMktBudget = totalMktBudget * combinedMktRate;
            var frontLoadEnd = 6 * tShift; var mktFrontLoad = (sku.stage === 'intro' || sku.stage === 'growth') ? 0.6 : 1.0; 

            for(var m2=1; m2<=months; m2++) {
                var sales = salesBase[m2-1] * multiplier; var mkt = 0;
                if(sku.stage === 'intro' || sku.stage === 'growth') {
                    if (m2 <= frontLoadEnd) mkt = (totalMktBudget * mktFrontLoad) / Math.max(1, frontLoadEnd); 
                    else mkt = (totalMktBudget * (1 - mktFrontLoad)) / Math.max(1, (months - frontLoadEnd)); 
                } else { mkt = sales * combinedMktRate; }
                cumulativeCash += (sales - (sales*COGS_RATE) - (sales*(sku.c/100)) - mkt);
                cashFlow.push(cumulativeCash);
            }
            return cashFlow;
        }

        function runMicroSimulation() {
            if (!currentMicroSkuId) return;
            var sku = null; for(var i=0; i<portfolio.length; i++) { if(portfolio[i].id===currentMicroSkuId) sku=portfolio[i]; }
            if(!sku) return;
            var tShift = parseFloat(document.getElementById('micro-timeshift').value); sku.timeShift = tShift;
            var timeLabel = document.getElementById('micro-time-label');
            if (tShift === 1.0) { timeLabel.innerText = '原计划执行 (100%耗时)'; timeLabel.className = "text-center mt-3 text-[14px] font-black text-slate-600"; }
            else if (tShift < 1.0) { timeLabel.innerText = '🔥 提前 ' + ((1-tShift)*100).toFixed(0) + '%'; timeLabel.className = "text-center mt-3 text-[15px] font-black text-emerald-600"; }
            else { timeLabel.innerText = '❄️ 延期 ' + ((tShift-1)*100).toFixed(0) + '%'; timeLabel.className = "text-center mt-3 text-[15px] font-black text-red-500"; }

            var throttle = parseInt(document.getElementById('global-throttle').value) / 100;
            var elastic = sku.stage === 'intro' ? 0.8 : (sku.stage === 'growth' ? 0.5 : (sku.stage === 'mat' ? 0.1 : 0));
            var actualRev = sku.actualVol * sku.price * (1 + ((throttle - 1) * elastic)); 
            var combinedMktRate = ((sku.ipo8 + sku.ipo9 + sku.ipo10) * throttle) / 100;

            var baseCashFlow = generateTimeSeries(sku, combinedMktRate, actualRev, 1.0);
            var actualCashFlow = generateTimeSeries(sku, combinedMktRate, actualRev, tShift);

            var baseBEP = -1, actualBEP = -1;
            for(var i=0; i<36; i++) { if(baseBEP===-1 && baseCashFlow[i]>=0) baseBEP = i+1; }
            for(var i=0; i<36; i++) { if(actualBEP===-1 && actualCashFlow[i]>=0) actualBEP = i+1; }
            var profitDiff = actualCashFlow[35] - baseCashFlow[35];

            var reportHtml = '';
            if (tShift < 1.0) { reportHtml += '<div class="bg-emerald-900/20 p-2 rounded border-l-2 border-emerald-500"><span class="text-emerald-600 font-bold">🎯 提前打透：</span><br>回本 <strong>提前至第 ' + (actualBEP>0?actualBEP:'∞') + ' 月</strong>。<br>3年累计<strong>增加 ¥' + profitDiff.toFixed(0) + '万</strong> 净利！</div>'; }
            else if (tShift > 1.0) { reportHtml += '<div class="bg-red-900/20 p-2 rounded border-l-2 border-red-500"><span class="text-red-500 font-bold">💀 延期反噬：</span><br>回本 <strong>延后至第 ' + (actualBEP>0?actualBEP:'∞') + ' 月</strong>。<br>利润<strong>缩水 ¥' + Math.abs(profitDiff).toFixed(0) + '万</strong>！</div>'; }
            else { reportHtml += '<div class="bg-slate-100 p-2 rounded text-slate-500">按既定时间表推进。拖拉上方滑块测算。</div>'; }
            document.getElementById('micro-report').innerHTML = reportHtml;

            if (typeof Chart !== 'undefined') {
                var masterLabels = []; for(var m=1; m<=36; m++) masterLabels.push('M'+m);
                safeDrawMicroChart(masterLabels, actualCashFlow, baseCashFlow);
            }
            runModel();
        }

        // ==================== 自由伸缩功能 ====================
        function initResizer() {
            var resizer = document.getElementById('drag-resizer');
            var topDeck = document.getElementById('top-deck');
            if(!resizer || !topDeck) return;
            var isResizing = false;

            var startResize = function(e) { isResizing = true; resizer.style.backgroundColor = '#94a3b8'; if(e.cancelable) e.preventDefault(); };
            var stopResize = function() { isResizing = false; resizer.style.backgroundColor = '#cbd5e1'; };
            var doResize = function(clientY) {
                if (!isResizing) return;
                var newHeight = (clientY / window.innerHeight) * 100;
                if (newHeight >= 25 && newHeight <= 75) topDeck.style.flex = '0 0 ' + newHeight + '%';
            };

            resizer.addEventListener('mousedown', startResize); resizer.addEventListener('touchstart', startResize, {passive: false});
            document.addEventListener('mousemove', function(e) { doResize(e.clientY); });
            document.addEventListener('touchmove', function(e) { if(isResizing && e.touches.length > 0) { doResize(e.touches[0].clientY); if(e.cancelable) e.preventDefault(); } }, {passive: false});
            document.addEventListener('mouseup', stopResize); document.addEventListener('touchend', stopResize);
        }
